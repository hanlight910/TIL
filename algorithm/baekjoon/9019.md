# 💳 문제이해

수 `a`가 주어집니다. 범위는 9,999이하 0이상입니다. 
`a`를 가지고 같은 조건의 수 `b`를 만들어야 합니다.
네가지 연산이 주어집니다.

1. D: `a`를 두배로 만듭니다. 9,999를 넘어간다면, 0으로 넘어가면 10,000으로 
나누고 나머지를 구합니다.
2. S: `a`에서 1을 뺍니다. 0 미만이면, 9999가 저장이 됩니다.
3. L: `a`의 각 자릿수를 왼쪽으로 회전시킵니다. 1,234 -> 2,341
4. R: `a`의 각 자릿수를 오른쪽으로 회전시킵니다. 1,234 -> 4,123

위 네가지 연산을 사용해서 `a`를 `b`로 만들기 위해 필요한 최소 연산 횟수와
그 연산을 과정을 구하시오.

## 🚥 문제접근

`BFS`로 목표가 나올 수 있는 경우의 수를 탐색하는게 좋을 것 같습니다.

우선, 연산 로직을 처리하는게 먼저인 것 같습니다.

### 🌏 분석

#### 💿 연산자

1. D연산자는 곱하기 2를 해주고 9999가 넘어가면 10,000의 나머지를 구하면 될 것
같습니다.
2. S연산자도 우선 0인 걸 화인하고, 0이면 9999로 저장하고, 아니면 1을 빼주면
될 것 같습니다.
3. L연산자는 문자열로 처리하면 간단하게 처리할 수 있습니다.
하지만 일반 정수로는 고민이 됩니다.
    - 1의자릿수가 4번째 자릿수로 와야 합니다. 이는 10의 나머지를 구하면
    간단하게 1의 자릿수를 구할 수 있습니다.
    - 1000의 자릿수 또한 1,000의 몫으로 구할 수 있습니다.
    - 100의 자릿수도 100의 몫으로 구할 수 있습니다.
    - 10의 자릿수도 10의 몫으로 구할 수 있습니다.
    - 1의 자릿수 * 1000 + 1000의 자릿수 * 100 + ...
4. 3번째 사항과 유사하게 처리하면 될 것 같습니다. 순서는 바꾸고.

#### 🌂 BFS

`BFS`를 통해 모든 연산한 뒤에 경우들을 탐색합니다.

`BFS`특성상 처음으로 방문한 경우가 시작점으로부터의 최단 거리입니다.

##### 🖥️ source code

```cpp
#include <cstdio>
#include<iostream>
#include<vector>
#include<queue>

#define FAST_IO ios_base::sync_with_stdio(false);\
	cin.tie(NULL)

using namespace std;
#define MAX_BOUND 9999
#define MIN_BOUND 0

enum OPERATION { D, S, L, R };

int32_t D_operation(int32_t a) {
    return a * 2 > MAX_BOUND ? a * 2 % (MAX_BOUND + 1) : a * 2;
}

int32_t S_operation(int32_t a) {
    return a > MIN_BOUND ? a - 1 : 9999;
}

int32_t L_operation(int32_t a) {
    int32_t first_digit = a % 10;
    int32_t second_digit = a % 100 / 10;
    int32_t third_digit = a % 1000 / 100;
    int32_t fourth_digit = a / 1000;
    return (third_digit * 1000) + (second_digit * 100) + (first_digit * 10) + (fourth_digit * 1);
}

int32_t R_operation(int32_t a) {
    int32_t first_digit = a % 10;
    int32_t second_digit = a % 100 / 10;
    int32_t third_digit = a % 1000 / 100;
    int32_t fourth_digit = a / 1000;
    return (first_digit * 1000) + (fourth_digit * 100) + (third_digit * 10) + (second_digit * 1);
}

int32_t operation(OPERATION o, int32_t a) {
    switch(OPERATION(o)) {
        case D:
            return D_operation(a);
        case S:
            return S_operation(a);
        case L:
            return L_operation(a);
        case R:
            return R_operation(a);
    }
    return -1;
}

typedef vector<int32_t> VecD;
struct op_idx {
	int32_t idx; char op;
};
/*typedef pair<int32_t, char> op_idx;*/

char enum_to_char(OPERATION idx) {
	switch(idx) {
		case D:
			return 'D';
		case S:
			return 'S';
		case L:
			return 'L';
		case R:
			return 'R';
	}
	return '\0';
}

vector<op_idx> bfs(int32_t a, int32_t target) {
	VecD visited(MAX_BOUND + 1, 0);
	/*VecD depth(MAX_BOUND + 1, 0);*/
	vector<op_idx> tracking(MAX_BOUND + 1, {-1, ' '});
	queue<int32_t> q;
	visited[a] = 1;
	q.push(a);

	while (!q.empty()) {
		int32_t cur_num = q.front();
		q.pop();	

		for (int32_t i = 0; i < 4; i += 1) {
			int32_t next_num = operation(OPERATION(i), cur_num);
			if (visited[next_num] == 0) {
				q.push(next_num);
				visited[next_num] = 1;
				tracking[next_num] = {
					cur_num,
					enum_to_char(OPERATION(i)),
				};
				/*depth[next_num] = depth[cur_num] + 1;*/
			}
		}
	}

	return tracking;

}

int32_t backtracking(vector<op_idx> tracked, int32_t target) {
	int32_t cur_idx = target;
	vector<char> result;
	while (cur_idx != -1) {
		result.push_back(tracked[cur_idx].op);
		cur_idx = tracked[cur_idx].idx;
	}

	for (int32_t i = 0; i < result.size() / 2; i += 1) {
		char temp = result[i];
		result[i] = result[result.size() - 2 - i];
		result[result.size() - 2 - i] = temp;
	}

	for (int32_t i = 0; i < result.size(); i += 1) {
		cout << result[i];
	}

	cout << '\n';
	return 0;
}

int32_t solution(int32_t a, int32_t b) {
	vector<op_idx> tracked = bfs(a, b);
	backtracking(tracked, b); 
	return 0; 
}

int32_t main(void) {
	FAST_IO;
	int32_t T;
	cin >> T;
	for (int32_t i = 0; i < T; i += 1) {
		int32_t a, b;
		cin >> a >> b;
		solution(a, b);
	}
    return 0;
}
```
