# 💳 문제이해

`N`개의 도시와, `M`개의 도시에서 도시 사이를 갈 수 있는 버스가 주어집니다.
시작 도시, 도착 도시 버스 번호가 주어지면 모든 도시에서 모든 도시에 갈 수 
있는 최소 경로와 그 최소 경로에서 거치는 도시를 구하시오.

## 🚥 문제접근

> "모든 도시의 본인을 포함한 모든 도시 사이의 최소 거리를 구합니다."

위 말은 `N x N`개의 쌍을 구한다는 의미이기도 합니다.

모든 도시가 시작점이므로 플로이드-워셜(Floyd-Warshall) 알고리즘을 사용할 수
있습니다. 이 경우 시간 복잡도는 $O(V^3)$입니다.

다익스트라 알고리즘은 단일 출발점에서 다른 모든 정점까지의 최단 거리를 찾는 데
최적화되어 있습니다. 일반적으로 $O((V + E) \log V)$ 시간 복잡도를 가지며, E는
간선의 수입니다. 그러나 모든 도시와 모든 도시 사이의 최단 거리를 구하려면 각
도시를 출발점으로 설정하고 다익스트라를 여러 번 실행해야 하므로 총 시간
복잡도는 $O(V(E + V \log V))$가 됩니다.

따라서 모든 도시 사이의 최단 거리를 찾는 경우, 플로이드-워셜 알고리즘이
$O(V^3)$의 시간 복잡도로 더 효율적일 수 있습니다. 특히 간선의 수 E가 $V^2$에
가까운 경우에는 플로이드-워셜이 더 유리합니다.

모든 쌍을 구해주는 플로이드 알고리즘의 시간복잡도는 $O(V^3)$입니다.
`floyd-warshall` 알고리즘이 더 효율적이므로 해당 알고리즘을 사용해봅니다.

### 🌏 분석

최단 경로를 구하는 알고리즘인 `floyd-warshall`은 모든 짝을 구해줍니다.
이는 행열 구조의 최단 경로를 저장하면서 이전에 있었던 최단 경로를 더 짧으면
갱신해 나가는 문제입니다.

타 최단 경로 구하는 알고리즘이랑 다르게 간선 개수가 의미가 없습니다. 
모든 도시 사이를 구할 거라서 그렇습니다.

#### 💾 Floyd-Warshall

행열 `Matrix`구조의 자료 구조를 사용합니다. 

**유의**:
1. 자기 자신과의 최단 경로는 0다.

$$dp[i][j] = min(dp[i][j], dp[i][k] + dp[k][j])$$

거죠 1부터 3까지의 최단 거리는 어떤 수부터 어떤 1 부터 어떤 수까지,
어떤 수부터 3까지이기도 합니다. 1부터 3까지 다이렉트로 갈 수 있는
경우를 제외하면 말이죠.

```cpp
#include <algorithm>
#include<iostream>
#include<vector>

#define INF 0x7FFFFFFF
using namespace std;

struct Edge {
    int32_t vert;
    int32_t weig;
};

class Graph {
    private:
        int32_t size;
        vector<vector<Edge>> adj_l;

    public:
        void init(int32_t size) {
            this->size = size;
            this->adj_l.resize(size + 1);
        }
		int32_t get_size() {
			return this->size + 1;
		}
        void push_edge(int32_t src, int32_t dest, int32_t weight) {
            this->adj_l[src].push_back({ dest, weight });
        }

		int32_t get_weight(int32_t src, int32_t dest) {
			return this->adj_l[src][dest].weig;
		}
		int32_t get_dest(int32_t src, int32_t dest) {
			return this->adj_l[src][dest].vert;
		}
		vector<Edge> get_adj_list(int32_t src) {
			return this->adj_l[src];
		}
};

vector<vector<int32_t>> floyd_warshall(Graph g) {
    vector<vector<int32_t>> dp(g.get_size(), vector<int32_t>(g.get_size(), INF));
	vector<vector<int32_t>> tracking(g.get_size(), vector<int32_t>(g.get_size(), -1));
    
    for (int32_t i = 1; i < g.get_size(); i += 1) {
        for (int32_t j = 0; j < g.get_adj_list(i).size(); j += 1) {
			if (dp[i][g.get_dest(i, j)] > g.get_weight(i, j)) {
				dp[i][g.get_dest(i, j)] = g.get_weight(i, j);
				tracking[i][g.get_dest(i, j)] = i;
			}
        }
    }


    for (int32_t i = 1; i < g.get_size(); i += 1) {
        dp[i][i] = 0; 
    }
	/*for (int32_t i = 1; i < g.get_size(); i += 1) {*/
	/*	for (int32_t j = 1; j < g.get_size(); j += 1) {*/
	/*		cout << dp[i][j] << ' ';*/
	/*	}*/
	/*	cout << '\n';*/
	/*}*/

    for (int32_t k = 1; k < g.get_size(); k += 1) {
        for (int32_t i = 1; i < g.get_size(); i += 1) {
            for (int32_t j = 1; j < g.get_size(); j += 1) {
                if (dp[i][j] > (int64_t)dp[i][k] + dp[k][j]) {
                    dp[i][j] = dp[i][k] + dp[k][j];
					tracking[i][j] = tracking[k][j];
                }
            }
        }
    }

	/*for (int32_t i = 1; i < g.get_size(); i += 1) {*/
	/*	for (int32_t j = 1; j < g.get_size(); j += 1) {*/
	/*		cout << dp[i][j] << ' ';*/
	/*	}*/
	/*	cout << '\n';*/
	/*}*/

    for (int32_t i = 1; i < g.get_size(); i += 1) {
        for (int32_t j = 1; j < g.get_size(); j += 1) {
            if (dp[i][j] >= INF) cout << '0' << ' ';
            else cout << dp[i][j] << ' ';
        }
        cout << '\n';
    }

    return tracking;
}

int32_t backtracking(int32_t size, vector<vector<int32_t>> tracked) {

	for (int32_t i = 1; i < size; i += 1) {
		for (int32_t j = 1; j < size; j += 1) {
			int32_t idx = j;

			vector<int32_t> result;

			while (idx != -1) {
				result.push_back(idx);
				idx = tracked[i][idx];
			}

			if (result.size() == 1) {
				cout << 0 << ' ';
			} else {
				
				reverse(result.begin(), result.end());
				cout << result.size() << ' ';

				for (int32_t i = 0; i < result.size(); i += 1) {
					cout << result[i] << ' ';
				}
			}
			
			cout << '\n';
		}
	}
	return 0;
}

int32_t solution(Graph& g) {
	vector<vector<int32_t>> tracked = floyd_warshall(g);
	backtracking(g.get_size(), tracked);
	return 0;
}

int32_t main(void) {
	int32_t N, M;
	cin >> N >> M;
	Graph g;
	g.init(N);

	for (int32_t i = 0; i < M; i += 1) {
		int32_t src, dest, weig;
		cin >> src >> dest >> weig;
		g.push_edge(src, dest, weig);
	}

	solution(g);

	return 0;
}
```
