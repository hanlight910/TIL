# 💳 문제이해

1부터 V까지의 도시가 주어집니다. 각 도시 사이에는 도로가 주어집니다.
도로는 E개 주어집니다. 각 도시 사이의 적은 사이클, 쌍이 되는 
도로 사이의 소요되는 시간의 최소 값을 구하세요. 

## 🚥 문제접근 

위 문제는 APSP(All-pairs shortest path) 모든 쌍 중에 최단 경로를 구합니다.
쌍을 구하는 알고리즘에서 잘 알려진 `Floyd-Warshall` 알고리즘을 사용합니다.

### 🌏 분석

#### 💾 Flowyd-Warshall

`Matrix` 자료 구조에 각 쌍의 최단 경로를 저장하는 알고리즘입니다.
각 자료 쌍을 구하는 문제이기 때문에, 시간 복자도는 $O(n^3)$ 라고 합니다.

`Dynamic Programming` 기법을 사용하면서 이전에 계산한 최단 경로에서
더 짧은 경로가 있으면 더 짧은 경로를 저장합니다.

각 정점에서 모든 정점으로 갈 수 있는 경로를 탐색하기 때문에, 시간 복잡도에
간선 개수는 필요하지 않습니다. 어차피 모두 탐색하기 때문입니다.

### 🛠️ 풀이

#### 🖥️ source code  

```cpp
#include<iostream>
#include<vector>
#include<stdio.h>

#define INF (0x7FFFFFFF)

struct Edge {
	int32_t v, w;
	Edge(int32_t v, int32_t w) {
		this->v = v;
		this->w = w;
	}
};

class Graph {
	public: 
		int32_t V;
		int32_t N;
		std::vector<std::vector<Edge>> v;

		Graph(int32_t v, int32_t n) {
		  this->V = v;
		  this->N = n;
		  this->v.resize(v);
		}
};

int32_t floyd_wayshall(Graph g) {
	std::vector<std::vector<int32_t>> dp(g.V, std::vector<int32_t>(g.V));

	for (int32_t i = 0; i < g.V; i += 1) {
		for (int32_t j = 0; j < g.V; j += 1) {
			if (i == j) dp[i][j] = 0;
			else dp[i][j] = INF;
		}
	}
	for (int32_t i = 1; i < g.V; i += 1) {
		for (int32_t j = 0; j < g.v[i].size(); j += 1) {
			dp[i][g.v[i][j].v] = g.v[i][j].w;
		}
	}

	for (int32_t i = 1; i < g.V; i += 1) {
		for (int32_t j = 1; j < g.V; j += 1) {
			for (int32_t k = 1; k < g.V; k += 1) {
				if (dp[i][j] > (int64_t)dp[i][k] + dp[k][j]) {
					dp[i][j] = dp[i][k] + dp[k][j];
				}
			}
		}
	}
	// for (int32_t i = 1; i < g.V; i += 1) {
	// 	for (int32_t j = 1; j < g.V; j += 1) {
	// 		if (dp[i][j] >= INF) printf("-1 ");
	// 		else printf("%2d ", dp[i][j]);
	// 	}
	// 	printf("\n");
	// }

	int32_t min = INF;

	for (int32_t i = 1; i < g.V; i += 1) {
		for (int32_t j = 1; j < g.V; j += 1) {
			if (i != j && min > (int64_t)dp[i][j] + dp[j][i]) {
				min = dp[i][j] + dp[j][i];
			}
		}
	}
	return min >= INF ? -1 : min;
}

int32_t main(void) {
	int32_t N, E;
	scanf("%d %d", &N, &E);
	Graph g = Graph((N + 1), E);

	for (int32_t i = 1; i <= E; i += 1) {
		int32_t u, v, w;
		scanf("%d %d %d", &u, &v, &w);
		g.v[u].push_back(Edge(v, w));
	}

	int32_t min_pair = floyd_wayshall(g);

	printf("%d\n", min_pair);
	return 0;
}
```
