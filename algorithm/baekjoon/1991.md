# 💳 문제이해

`binary tree`(이진 나무)가 주어지면 세가지 `traverse`(순회)를 사용해서
노드를 출력합니다.

1. 전위순회(preorder traverse): root->left->right
2. 중위수회(inorder traverse): left->root->right
3. 후의순회(postorder traverse): left->right->root


# 🚥 문제접근

`binary tree` 순환 방법입니다.
자료 구조는 `left`,`right`, `parent` 노드가 있어야 합니다.

## 🌏 분석

1. 전위순회 
    - 루트에서 왼쪽 선, 오른쪽 후
    - 위에서 아래로
    - `DFS`사용
2. 중위순회
    - 왼쪽 우선 탐색
    - 부모 탐색
    - 오른쪽 탐색
3. Postorder traverse
    - 왼쪽 우선 탐색
    - 오른쪽 탐색
    - 부모 탐색


### 🛠️ 풀이

#### 🖥️ source code

```cpp
#include<iostream>
#include<vector>
#include<stack>

using namespace std;

struct Node {
	int32_t left;
    int32_t right;
    int32_t parent;
	char val;
};

int32_t preorder_traverse(vector<Node> v, int32_t start) {
    stack<Node> st;
	st.push(v[start]);
	
	while (!st.empty()) {
		Node cur_n = st.top();
		st.pop();
		cout << cur_n.val;
		if (cur_n.right != -1) {
			st.push(v[cur_n.right]);
		}
		if (cur_n.left != -1) {
			st.push(v[cur_n.left]);
		}
	}
	cout << '\n';
    return 0;
}

int32_t inorder_traverse(vector<Node>& v, int32_t src) {
    stack<int32_t> s;
    while (src != -1 || !s.empty()) {
        while (src != -1) {
            s.push(src);
            src = v[src].left;
        }
        src = s.top();
        s.pop();
        cout << v[src].val;
        src = v[src].right;
    }
    return 0;
}

int32_t postorder_traverse(vector<Node>& v, int32_t src) {
    stack<int32_t> s1, s2;
    s1.push(src);
    
    while (!s1.empty()) {
        int32_t node = s1.top();
        s1.pop();
        s2.push(node);
        
        if (v[node].left != -1) {
            s1.push(v[node].left);
        }
        if (v[node].right != -1) {
            s1.push(v[node].right);
        }
    }
    
    while (!s2.empty()) {
        cout << v[s2.top()].val;
        s2.pop();
    }
    return 0;
}

int32_t main(void) {
	int32_t N;
	cin >> N;
	vector<Node> v(N);

	for (int32_t i = 0; i < N; i += 1) {
		char parent, left, right;
		cin >> parent >> left >> right;
		v[parent - 'A'].left = -1;
		v[parent - 'A'].right = -1;
		if (left != '.') {
			v[parent - 'A'].left = left - 'A';
			v[left - 'A'].parent = parent;
		}
		if (right != '.') {
			v[parent - 'A'].right = right - 'A';
			v[right - 'A'].parent = parent;
		}
		v[parent - 'A'].val = parent;
	}

	v[0].parent = -1;

	preorder_traverse(v, 0);
	inorder_traverse(v, 0);
	cout << '\n';
	postorder_traverse(v, 0);
	cout << '\n';
    return 0;
} 
```
