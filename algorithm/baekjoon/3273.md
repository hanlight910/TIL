# 💳 문제이해

N개의 수열이 주어집니다. 각 원소는 1보다 크거나 같으며, 1,000,000 작거나
같습니다. 정수 x가 주어졌을 때, 서로 다른 원소를 더하였을 x가 되는
쌍의 개수를 구하세요.

## 🚥 문제접근

정확히 sigma N횟수를 돌아 각 각원소를 모든 원소와 비교하는 방법이
있습니다. 하지만 그 방법은 비효율적입니다. 더 효율적인 방법이 있습니다.

x값에서 각 원소를 뺀값이 그 원소의 짝입니다. 

### ⚒️  풀이

우선 각 원소의 짝을 찾습니다.
짝을 찾았다면, 그 짝이 주어진 원소들의 있는지 확인합니다.
있으면 개수를 더합니다.

다 더한 개수가 서로 다른 두 원소의 합이 x인 짝입니다.

#include <algorithm>

#define MAX_ELEMENT 2000001

int32_t solution(std::vector<int32_t> inputs, int32_t n, int32_t x) {
    int32_t* pairwise = (int32_t*)calloc(MAX_ELEMENT, sizeof(int32_t));
    int32_t count = 0;
	int32_t i = 0, j = n - 1;

	// while (i < j) {
	// 	if (inputs[i] + inputs[j] == x) {
	// 		i += 1;
	// 		j -= 1;
	// 		count += 1;intpu
	// 	} else if (inputs[i] + inputs[j] > x) {
	// 		j -= 1;	
	// 	} else {
	// 		i += 1;
	// 	}
	// }
	for (int32_t i = 0; i < n; i += 1) {
		if (x - inputs[i] < 0) {
			continue;
		}
		pairwise[x - inputs[i]] += 1;
		// printf("pair: %d\n", inputs[i]);
	}

	for (int32_t i = 0; i < n; i += 1) {
		if (pairwise[inputs[i]] > 0) {
			count += 1;
		}
	}
    return count / 2;
}

int32_t main(void) {
    int32_t N, x;
    scanf("%d", &N);
    std::vector<int32_t> elements(N);
    for (int32_t i = 0; i < N; i += 1) {
        scanf("%d", &elements[i]);
    }
	// std::sort(elements.begin(), elements.end());
	scanf("%d", &x);
    int32_t count_pairwise = solution(elements, N, x);
    printf("%d\n", count_pairwise);

    return 0;
}
