# 💳 문제이해

`N`개의 도시가 주어집니다. 각 도시 사이에 운행하는 버스가 있습니다.
버스는 출발하는 도시, 도착하는 도시 소요되는 비용이 주어집니다.
주어진 시작 도시부터 도착 도시까지의 최소 비용과, 거치는 도시의 수와
그 도시들을 출력합니다.

## 🚥 문제접근

위 문제는 가중치가 존재하는 그래프에서 주어진 간선들을 
토대로 특정 정점과 정점 사이의 최단 거리를 구하는 문제이면서, 
해당 도시들을 구하는 문제입니다.

가중치가 설로 동일하다면 `BFS`방법을 사용할 수 있겠지만, 
가중치는 0이상의 정수입니다. 가중치가 0을 포함한 양의 정수인 경우
`Dijkstra` 알고리즘을 사용해 최단 경로를 구할 수 있습니다.

최단 경로와 별개로 최단 경로를 사용하는 경로의 거처야 하는 도시들은
개별적으로 처리해야 합니다. 

### 🌏 분석

`Dijkstra` 알고리즘 가중치가 음의 정수가 아닌 그래프에서
최단 경로를 구하기 적절합니다.

#### 🛠️ 풀이

##### 🖥️ source code 

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <limits>

using namespace std;

struct Edge {
	int32_t dest;
	int32_t weight;
};

struct CompareEdge {
	bool operator()(const Edge& e1, const Edge& e2) const {
		return e1.weight > e2.weight;
	}
};

class Graph {
	public:
		explicit Graph(int32_t size) : size(size), adjacent_list(size + 1) {}

		void addEdge(int32_t src, int32_t dest, int32_t weight) {
			adjacent_list[src].emplace_back(Edge{dest, weight});
		}

		const vector<Edge>& neighbors(int32_t vertex) const {
			return adjacent_list[vertex];
		}

		int32_t getSize() const {
			return size;
		}

	private:
		vector<vector<Edge>> adjacent_list;
		int32_t size;
};

vector<int32_t> dijkstra(const Graph& g, int32_t start, int32_t target) {
	priority_queue<Edge, vector<Edge>, CompareEdge> pq;
	vector<int32_t> tracking(g.getSize() + 1, -1);
	vector<int64_t> distance(g.getSize() + 1, numeric_limits<int64_t>::max());
	vector<bool> visited(g.getSize() + 1, false);

	pq.push({start, 0});
	distance[start] = 0;

	while (!pq.empty()) {
		Edge cur_e = pq.top();
		pq.pop();

		if (cur_e.weight > distance[cur_e.dest]) 
			continue;

		for (const Edge& neighbor_e : g.neighbors(cur_e.dest)) {
			int32_t neighbor_vertex = neighbor_e.dest;
			int32_t new_distance = distance[cur_e.dest] + neighbor_e.weight;


			if (new_distance < distance[neighbor_vertex]) {
				distance[neighbor_vertex] = new_distance;
				pq.push({neighbor_vertex, new_distance});
				tracking[neighbor_vertex] = cur_e.dest;
			}
		}
	}

	cout << distance[target] << '\n';

	return tracking;
}

int32_t backtracking(vector<int32_t> tracked, int32_t target) {
	vector<int32_t> result;

	int32_t idx = target;
	
	while (idx != -1) {
		result.push_back(idx);
		idx = tracked[idx];
	}

	cout << result.size() << '\n';
	for (int32_t i = 0; i < result.size() / 2; i += 1) {
		int32_t temp = result[i];
		result[i] = result[result.size() - 1 - i];
		result[result.size() - 1 - i] = temp;
	}

	for (auto& el: result) {
		cout << el << ' ';
	}
	cout << '\n';

	return 0;
}

int32_t solution(const Graph& g, int32_t start, int32_t target) {
	vector<int32_t> tracked = dijkstra(g, start, target);
	backtracking(tracked, target);
	return tracked[target];
}

int32_t main() {
	int32_t N, num_edges;
	cin >> N >> num_edges;

	Graph g(N);

	for (int32_t i = 0; i < num_edges; ++i) {
		int32_t src, dest, weight;
		cin >> src >> dest >> weight;
		g.addEdge(src, dest, weight);
	}

	int32_t start, target;
	cin >> start >> target;

	solution(g, start, target);

	return 0;
}
```
