# 💳 문제이해

1부터 N개의 정점이 주어집니다. 간선도 E개 주어집니다. 간선은 두개의 정점과 가중치가 
주어집니다.

1부터 N사이의 최단 거리를 구합니다. 
단 조건이 있습니다. 간선이 주어진 다음에 최단 거리의 사이에 무조건 방문해야 하는
두 정점이 주어집니다. 두 정점은 서로 같지 않으며, 두 정점의 간선은 최대 1개가 
주어집니다. 조건을 만족하면서 도달할 수 없다면 -1를 출력합니다.

## 🚥 문제접근

적어도 목적직에 도달을 못한다 하더라도 조건을 만족해야 합니다.

우선적으로 조건 정점을 방문하는 것은 최단 경로가 아닐 가능성도 있기 때문에,
피합니다. 각 구간의 최단 경로를 우선 구한 다음에, 1부터 v1, v2 N까지 사이의
최단 경로를 계산하면 될 것 같습니다.

가중치가 양의 정수이면서 서로 다른 경우 `Daijkstra` 알고리즘을 사용할 수 있습니다.

### Daijkstra(다익스트라)

가중치가 서로 동일 한 경우 m 부터 n까지의 최단 경로를 구할 수 있는 알고리즘입니다.

`binary heap`를 사용하는 경우,
시간 복잡도는 $O((V + E)log(V))$ 입니다.

### 🛠️ 풀이

##### source code

```c
#include<stdio.h>
#include<stdint.h>
#include<stdlib.h>

#define MAX_WEIGHT 2400001
#define MAX_EDGE 200001
#define DEFINE_DATA(identifier, w, n) Data identifier = { .weight = w, .num = n }

typedef struct {
    int32_t num;
    int32_t weight;
} Data;

typedef struct {
    Data* arr;
    int32_t size;
    int32_t capacity;
    int32_t top;
} heap;

typedef struct Node {
    Data val;
    struct Node* next;
} Node;

typedef struct Graph {
    int32_t size;
    Node** adjacent_list; 
} Graph;

heap* create_heap(int32_t length) {
    heap* n_h = (heap*)malloc(sizeof(heap));
    n_h->top = 0;
    n_h->size = 0;
    n_h->capacity = length;
    n_h->arr = (Data*)malloc(sizeof(Data) * length);
    return n_h;
}

void swap(Data* a, Data* b) {
    Data temp = *a;
    *a = *b;
    *b = temp;
    return;
}

void heapifyup(heap* h) {
    int32_t cur = h->size - 1;
    while (cur >= 0 && h->arr[cur].weight < h->arr[(cur - 1) / 2].weight) {
        swap(&h->arr[cur], &h->arr[(cur - 1) / 2]);
    }
    return;
}

void insert_heap(heap** h, Data val) {
    heap* temp_h = *h;
    if (temp_h->size >= temp_h->capacity) {
        printf("heap is already full!\n");
        return;
    }
    temp_h->arr[temp_h->size] = val;
    temp_h->size += 1;
    heapifyup(temp_h);
}

void heapifydown(heap* h) {
    int32_t cur = h->top;

    while (cur < h->size) {
        int32_t min = cur;
        if (cur * 2 + 1 < h->size && h->arr[cur * 2 + 1].weight < h->arr[min].weight) {
            min = cur * 2 + 1; 
        }
        if (cur * 2 + 2 < h->size && h->arr[cur * 2 + 2].weight < h->arr[min].weight) {
            min = cur * 2 + 2; 
        }
        if (cur != min) {
            swap(&h->arr[min], &h->arr[cur]);
        } else {
            break;
        }
    }
    return;
}

Data remove_heap(heap** h) {
    heap* temp  = *h;
    Data min = { -1, -1 };
    if (temp->size == 0) {
        printf("no elements\n");
        return min;
    }
    min = temp->arr[temp->top];
    temp->size -= 1;
    swap(&temp->arr[temp->top], &temp->arr[temp->size]);
    heapifydown(temp);
    return min;
}

Graph* create_graph(int32_t N) {
    Graph* n_g = (Graph*)malloc(sizeof(Graph)); 
    n_g->size = N;
    n_g->adjacent_list = (Node**)malloc(N * sizeof(Node*));
    for (int32_t i = 0; i < N; i += 1) {
        *((*n_g).adjacent_list + i) = NULL;
    }
    return n_g;
}

void add_edge(Graph* g, int32_t u, int32_t v, int32_t w) {
    Node* adj = g->adjacent_list[u];
    Node* new_v = (Node*)malloc(sizeof(Node));
    Data d = { .weight = w, .num = v };
    new_v->val = d;
    new_v->next = g->adjacent_list[u];
    g->adjacent_list[u] = new_v;

	Node* new_u = (Node*)malloc(sizeof(Node));
	DEFINE_DATA(d_v, w, u);
	new_u->next = g->adjacent_list[v];
	new_u->val = d_v;
	g->adjacent_list[v] = new_u;
    return;
} 

int32_t* daijkstra(Graph* g, int32_t start) {
    heap* h = create_heap(MAX_EDGE);
    DEFINE_DATA(temp, 0, start);

    insert_heap(&h, temp);
    int32_t* distance = (int32_t*)calloc(g->size, sizeof(int32_t));
    for (int32_t i = 0; i < g->size; i += 1) {
        distance[i] = MAX_WEIGHT;
    }
    distance[start] = 0;
    
    while (h->size > 0) {
        Data cur = remove_heap(&h);  
        Node* adj = *(g->adjacent_list + cur.num);

        while (adj != NULL) {
            if (distance[adj->val.num] > distance[cur.num] + adj->val.weight) {
                distance[adj->val.num] = distance[cur.num] + adj->val.weight;
                insert_heap(&h, adj->val);
            }
            adj = adj->next;
        }
    }

	
	return distance;
}
int32_t solution(Graph* g, int32_t v1, int32_t v2) {
	int32_t min_path = -1;
    int32_t* distance_from_one = daijkstra(g, 1);
	int32_t* distance_from_v1 = daijkstra(g, v1);
	int32_t* distance_from_v2 = daijkstra(g, v2);

	// 1 -> v1 -> v2 -> N
	int32_t first_path = distance_from_one[v1] + distance_from_v1[v2] + distance_from_v2[g->size - 1];
	// 1 -> v2 -> v1 -> N
	int32_t second_path = distance_from_one[v2] + distance_from_v2[v1] + distance_from_v1[g->size - 1];
	// printf("first path: %d\n", first_path);
	// printf("second path: %d\n", second_path);

	min_path = first_path > second_path ? second_path : first_path;
	min_path = min_path >= MAX_WEIGHT ? -1 : min_path;

	return min_path;
}

int32_t main(void) {
    int32_t V, E;
    scanf("%d %d", &V, &E);
    Graph* g = create_graph((V + 1));
    for (int32_t i = 0; i < E; i += 1) {
        int32_t u, v, w;
        scanf("%d %d %d", &u, &v, &w);
        add_edge(g, u, v, w);
    } 
	int32_t v1, v2;
	scanf("%d %d", &v1, &v2);
	int32_t shortest_path = solution(g, v1, v2);
	printf("%d\n", shortest_path);
    return 0;
}
```
